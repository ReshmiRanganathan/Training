16 Customer name and products they ordered
db.orders.aggregate([
  {
    $lookup: {
      from: "customers",
      localField: "customer_id",
      foreignField: "customer_id",
      as: "customer"
    }
  },
  { $unwind: "$customer" },
  {
    $lookup: {
      from: "products",
      localField: "product_id",
      foreignField: "product_id",
      as: "product"
    }
  },
  { $unwind: "$product" },
  {
    $project: {
      _id: 0,
      customer_name: "$customer.name",
      product_name: "$product.name"
    }
  }
])
17 Product name and total quantity sold

ADVANCED AGGREGATION:
18 Total revenue per product
db.orders.aggregate([
  {
    $lookup: {
      from: "products",
      localField: "product_id",
      foreignField: "product_id",
      as: "product"
    }
  },
  { $unwind: "$product" },
  {
    $group: {
      _id: "$product.name",
      totalRevenue: { $sum: { $multiply: ["$quantity", "$product.price"] } }
    }
  }
])
